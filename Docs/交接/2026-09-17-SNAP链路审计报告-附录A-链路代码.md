```csharp
private void RunOwnerPrediction()
{
    if (_locomotor == null) return;
    double deltaTime = TestPredictionDeltaTime >= 0f ? TestPredictionDeltaTime : Time.deltaTime;
    int steps = _accumulator.Advance(deltaTime, _fixedDelta, MovementPredictionConfig.ClientMaxCatchUpSteps);
    _diag.NoteGenerated(steps);
    for (int i = 0; i < steps; i++)
    {
        uint tick = ++_localTick;
        bool jump = _input != null && _input.JumpQueued;
        var cmd = new MovementCommand(
            _input != null ? _input.Move : Vector2.zero,
            _input != null && _input.Sprint,
            jump,
            _pendingYaw, _pendingPitch, tick);
        if (jump && _input != null) _input.ConsumeJump();
        _pendingYaw = 0f;
        _pendingPitch = 0f;

        _buffer.TryStore(cmd);
        Vector3 moveBefore = transform.position;
        _locomotor.Simulate(cmd, _fixedDelta);
        Vector3 moveAfterSimulate = transform.position;   // MoveAfter 紧接 Simulate 采样

        // C1：视角超前量扣减基准 = 本 tick 提交的输入（cmd.YawDelta）
        if (MovementPredictionConfig.ViewYawInterpolationEnabled)
            _viewYawDegrees = ConsumeViewYawLead(_viewYawDegrees, cmd.YawDelta);

        // 小误差平滑收敛：逐模拟步消费衰减校正偏移（大误差走 HardSnapTo）
        Vector3 appliedSmooth = Vector3.zero;
        if (_smoothActive)
        {
            Vector3 step = Reconciler.SmoothStep(_smoothOffsetRemaining, _fixedDelta);
            _smoothOffsetRemaining -= step;
            if (_smoothOffsetRemaining.sqrMagnitude < 1e-8f) { _smoothActive = false; _smoothOffsetRemaining = Vector3.zero; }
            _locomotor.ApplySmoothCorrection(step);
            appliedSmooth = step;
            if (!_correctionLedger.Record(tick, step))
            {
                // 台账溢出 = ACK 长时间未推进：在途量已丢失，平滑不可信——停用并等硬重基
                _smoothActive = false;
                _smoothOffsetRemaining = Vector3.zero;
            }
            _diag.NoteSmoothApplied(step.magnitude);
        }
        // C2：插值端点**逐步**更新（…注释略…）
        _interpFrom = moveBefore; _interpTo = transform.position; _interpValid = true;

        _buffer.RecordPredicted(_locomotor.CaptureSnapshot());   // 记录预测快照（键 = snapshot.Tick = cmd.Tick）
        CaptureClientStepEvidence(cmd, transform.position, moveAfterSimulate, appliedSmooth, null, moveBefore);
        RecordPullbackStep(cmd, moveBefore, moveAfterSimulate, transform.position, appliedSmooth);
        PushNextPredictedStep();   // 环形采样（同 tick 模拟状态）

        if (++_ticksSinceSend >= MovementPredictionConfig.ClientBatchSendEveryTicks)
        {
            _ticksSinceSend = 0;
            SendUnackedCommands();
        }
    }
}
```

```csharp
private void ServerTick()
{
    if (_locomotor == null) return;

    // Host 调试（Owner 在服务器本地）：从同一输入累积量采样命令入队——与远端玩家完全同一条权威管线
    if (IsOwner && _input != null && !LocalMovementFrozen)
    {
        bool hostJump = _input.JumpQueued;
        var hostCmd = new MovementCommand(_input.Move, _input.Sprint, hostJump, _pendingYaw, _pendingPitch, ++_hostSourceTick);
        if (hostJump) _input.ConsumeJump();
        _pendingYaw = 0f; _pendingPitch = 0f;
        _serverQueue.Enqueue(hostCmd);
    }

    if (MovementFrozen)
    {
        // 死亡/倒计时冻结：丢弃待处理输入（防解冻后爆发重放），权威快照照发（客户端对位）
        _serverQueue.Clear();
    }
    else
    {
        // One server tick advances the locomotor exactly once. Backlog is consumed
        // over subsequent server ticks; never multiply simulation time during catch-up.
        int count = _serverQueue.Drain(1, _serverBatch);          // ← 每 tick 恒消费 1 条（ServerMaxCatchUpPerTick=3 未用于此调用点）
        _diag.NoteServerConsumed(count);                          // 0 = 本 tick 空步
        if (_serverQueue.LastDrainGap > 0 && Time.unscaledTime >= _nextGapLogTime)
        {
            _nextGapLogTime = Time.unscaledTime + 1f;
            Debug.LogWarning($"[Day3][Input] GAP skipped={_serverQueue.LastDrainGap} total={_serverQueue.SkippedTicks} lastTick={_serverQueue.LastProcessedTick} qdepth={_serverQueue.Count} — 缺口视为跳过并结算（未被模拟），由客户端预测纠偏吸收");
        }
        if (count == 0)
        {
            // §3.2-2：空队列**不得**用空输入模拟（地面减速度会把权威角色刹停）→ 按最后已知输入有界外推
            var decision = ServerInputExtrapolation.Decide(_serverQueue, out var extrapolated);
            if (decision == ServerInputExtrapolation.Decision.Extrapolated)
            {
                _locomotor.Simulate(extrapolated, _fixedDelta);
                _diag.NoteExtrapolatedStep();
                CaptureServerStepEvidence(extrapolated, "EXTRAP");
                if (_serverQueue.ConsecutiveEmptyTicks == 1) ExportServerStepTraceWindow("IDLE_START");
            }
            else
            {
                _locomotor.Simulate(default, _fixedDelta);
                CaptureServerStepEvidence(default, "EMPTY");
            }
        }
        else
        {
            for (int i = 0; i < count; i++)
            {
                _locomotor.Simulate(_serverBatch[i], _fixedDelta);
                if (!IsOwner) ApplyRemotePitch(_serverBatch[i].PitchDelta);
                CaptureServerStepEvidence(_serverBatch[i], null);
            }
        }
    }

    // 每 tick 向 Owner 推权威快照（Host Owner 本身即权威，不发）
    var owner = NetworkObject != null ? NetworkObject.Owner : null;
    if (!IsOwner && owner != null && owner.IsValid)
        TargetAuthoritativeState(owner, BuildAuthoritativeState());

    EmitMovementDiagnostics("server");
}

private AuthoritativeMovementState BuildAuthoritativeState()
{
    var snapshot = _locomotor != null ? _locomotor.CaptureSnapshot() : default;
    return new AuthoritativeMovementState
    {
        ServerTick = TimeManager != null ? TimeManager.Tick : 0,
        LastClientTick = _serverQueue.LastProcessedTick,
        IdleStepsAtSnapshot = _serverQueue.ConsecutiveEmptyTicks,
        Dead = _combatAuthority != null && _combatAuthority.IsDead,
        Snapshot = snapshot,
    };
}
```

```csharp
private void ApplyOwnerAuthoritativeState(AuthoritativeMovementState state)
{
    _lastServerTick = state.ServerTick;
    _lastIdleStepsAtSnapshot = state.IdleStepsAtSnapshot;
    _diag.AuthoritativePitchDegrees = state.Snapshot.Pitch;

    if (state.Dead)                       // ① 死亡边界：无条件处理（重复/乱序快照不拦）
    {
        bool firstDeathSnapshot = !_ownerWasDead;
        _ownerWasDead = true;
        if (firstDeathSnapshot) _inputEpoch++;
        _gate.Observe(state.LastClientTick);
        _lastAckedClientTick = _gate.LastAckedClientTick;
        _divergenceGate.Reset();
        _buffer.Clear();
        _correctionLedger.Invalidate();
        _smoothActive = false; _smoothOffsetRemaining = Vector3.zero;
        _locomotor?.ApplyAuthoritativeSnapshot(state.Snapshot);
        _localTick = state.LastClientTick;
        if (firstDeathSnapshot) { _snapCount++; _diag.NoteRebase(MovementRebaseKind.DeathRespawn); }
        ResetOwnerVisualInterpolation();
        return;
    }

    if (_ownerWasDead)                    // ② 重生边界
    {
        _ownerWasDead = false;
        _inputEpoch++;
        _buffer.Clear();
        _divergenceGate.Reset();
        _gate.Observe(state.LastClientTick);
        _lastAckedClientTick = state.LastClientTick;
        _localTickAtLastAck = state.LastClientTick;
        HardSnapTo(state, MovementRebaseKind.DeathRespawn);
        ApplyOwnerRespawnPitchBaseline();
        return;
    }

    if (!_predictionAligned && state.LastClientTick != 0)   // ③ 首个快照：入场期领先一次性收敛（lead>20 才动）
    {
        _predictionAligned = true;
        int startupLead = _localTick > state.LastClientTick ? (int)(_localTick - state.LastClientTick) : 0;
        if (startupLead > MovementPredictionConfig.ClientLeadWarnTicks)
        {
            Debug.LogWarning($"[Day3][Input] LEAD_ALIGN local={_localTick} serverAck={state.LastClientTick} lead={startupLead} …");
            _gate.Observe(state.LastClientTick);
            _lastAckedClientTick = state.LastClientTick;
            _localTickAtLastAck = state.LastClientTick;
            _buffer.Clear(); _divergenceGate.Reset();
            HardSnapTo(state, MovementRebaseKind.Initial);
            return;
        }
    }

    switch (_gate.Feed(state.LastClientTick))    // ④ ACK 门：重复/乱序忽略；首次 InitialSnap；前进才比较
    {
        case ReconcileGateAction.IgnoreRepeat: return;
        case ReconcileGateAction.InitialSnap:  HardSnapTo(state, MovementRebaseKind.Initial); return;
        case ReconcileGateAction.Compare:
            _lastAckedClientTick = _gate.LastAckedClientTick;
            _localTickAtLastAck = _localTick;
            break;
    }

    // 必须在剪枝前读取 ACK 对应快照
    if (!_buffer.TryGetPredictedAndPrune(state.LastClientTick, out var predictedAtAck))
    {
        HardSnapTo(state, MovementRebaseKind.HistoryMissing);   // ⑤ 无配对历史（容量淘汰）→ 保守硬对位
        return;
    }

    var decision = Reconciler.Decide(state.Snapshot, predictedAtAck, out var correction);   // ⑥ 同 tick 配对比较（IdleStepsAtSnapshot 未参与）
    if (decision.Action == ReconcileAction.Snap)
    {
        Debug.Log($"[Day3][Reconcile] SNAP err={correction.magnitude:F3}m serverTick={state.ServerTick} ack={state.LastClientTick} totalSnaps={_snapCount + 1}");
        HardSnapTo(state, MovementRebaseKind.Snap);              // ← 本局 47 次全部走这里
        return;
    }
    if (_divergenceGate.Feed(decision))                          // ⑦ 软分叉滞回确认（yaw 2 次 / 其它 3 次）
    {
        _smoothActive = false; _smoothOffsetRemaining = Vector3.zero;
        Debug.Log($"[Day3][Reconcile] DIVERGENCE {decision.Divergence} serverTick={state.ServerTick} ack={state.LastClientTick} err={correction.magnitude:F3}m totalSnaps={_snapCount + 1}");
        HardSnapTo(state, MovementRebaseKind.Divergence);
        return;
    }
    if (_correctionLedger.Overflowed)                            // ⑧ 台账溢出 → 硬重基
    {
        Debug.LogWarning($"[Day3][Reconcile] LEDGER_OVERFLOW_REBASE ack={state.LastClientTick} serverTick={state.ServerTick} totalSnaps={_snapCount + 1}");
        HardSnapTo(state, MovementRebaseKind.Overflow);
        return;
    }

    // ⑨ 位置误差统一重算：新误差 = 服务器配对快照 − 客户端配对快照，扣除"配对快照之后已施加"的在途步进
    Vector3 inFlight = _correctionLedger.ConsumeInFlightAfter(state.LastClientTick);
    Vector3 corrected = correction - inFlight;
    _diag.LastErrorMeters = correction.magnitude;
    const float smoothThreshold = MovementPredictionConfig.ReconcileSmoothMeters;
    if (corrected.sqrMagnitude <= smoothThreshold * smoothThreshold)
    {
        _smoothActive = false; _smoothOffsetRemaining = Vector3.zero; _diag.PendingMeters = 0f;
    }
    else
    {
        _smoothOffsetRemaining = corrected; _smoothActive = true;
        _diag.NoteSmooth(corrected.magnitude, correction.magnitude);
    }
}
```

### 5.4.2 `HardSnapTo`（对位 + 重放未确认输入）— `PlayerNetworkAdapter.cs:970`

```csharp
private void HardSnapTo(AuthoritativeMovementState state, MovementRebaseKind kind)
{
    Vector3 previousVisualWorld = _viewOffsetRoot != null ? _viewOffsetRoot.position : transform.position;
    Vector3 rootBefore = transform.position;
    _locomotor?.ApplyAuthoritativeSnapshot(state.Snapshot);        // ← 对位：根/CC/速度/计时/债务全量恢复
    Vector3 rootAfterSnapshot = transform.position;
    _smoothActive = false; _smoothOffsetRemaining = Vector3.zero;
    _correctionLedger.Invalidate();                               // 绝对对位覆盖一切在途步进

    _snapCount++;
    _diag.NoteRebase(kind);
    Debug.Log($"[Day3][Reconcile] REBASE {kind} ack={state.LastClientTick} serverTick={state.ServerTick} totalRebases={_snapCount} idleSteps={state.IdleStepsAtSnapshot}");

    _traceSnapApplied = true; _traceSnapFrom = rootBefore; _traceSnapTo = rootAfterSnapshot;
    _traceRebaseKind = (int)kind; _traceRebaseError = Vector3.Distance(rootBefore, rootAfterSnapshot);

    uint replayed = state.LastClientTick;
    if (state.LastClientTick != 0 && _buffer.HasCommands)
    {
        uint newest = _buffer.NewestTick;
        for (uint tick = state.LastClientTick + 1; tick <= newest; tick++)
        {
            if (!_buffer.TryGetCommand(tick, out var cmd)) break;   // 缺口：命令不可得 → 停止重放
            Vector3 replayBefore = transform.position;
            _locomotor?.Simulate(cmd, _fixedDelta);
            _buffer.RecordPredicted(_locomotor.CaptureSnapshot());
            CaptureClientStepEvidence(cmd, transform.position, transform.position, Vector3.zero, "REPLAY", replayBefore);
            replayed = tick;
        }
        if (replayed < newest)
        {
            _buffer.InvalidatePredictedAbove(replayed);
            _diag.NoteReplayGap((int)(newest - replayed));
            Debug.LogWarning($"[Day3][Reconcile] REPLAY_GAP ack={state.LastClientTick} replayed={replayed} newest={newest} invalidated={newest - replayed}");
        }
        // tick 基线取 max(重放到的 tick, 缓冲最新 tick)：不回退（避免 TryStore 静默拒绝）、也不假装模拟了缺口
        _localTick = replayed > newest ? replayed : newest;
    }
    else
    {
        _localTick = state.LastClientTick;
    }
    _traceSnapTo = transform.position;                              // 重放结束后的最终根

    if (kind == MovementRebaseKind.DeathRespawn) ResetOwnerVisualInterpolation();
    else CarryVisualAfterRebase(previousVisualWorld);               // C3：普通纠偏保留视觉残差衰减

    ExportStepTraceWindow($"REBASE_{kind}", state.LastClientTick);  // 双端同 tick 小窗口取证
}
```
